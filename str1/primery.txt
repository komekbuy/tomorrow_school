================================================================================
КОНСПЕКТ — 6 ПРАВИЛ
================================================================================

1. Строка — это БАЙТЫ, а не буквы.
   len("Привет") = 12, а не 6.

2. for i, c := range s
   c — целая буква (руна)
   i — номер БАЙТА, с которого буква начинается (не номер буквы!)

3. s[i] даёт БАЙТ (число), а не букву.
   fmt.Println("Hello"[0])  ->  72
   На кириллице обращение по индексу ломается.

4. Строку менять нельзя. s[0] = 'H' не соберётся.
   Надо: []rune(s) -> поправить -> string(...) обратно.

5. 'a' — руна (число), "a" — строка.
   fmt.Print('a') печатает 97.

6. Склейка — просто +. Пустая строка "" — стартовое значение накопителя.

ПОРЯДОК В range: первое значение — номер, второе — буква.
   Имена любые (i, c / nomer, bukva / x, y) — важно МЕСТО, не имя.
   for c, i := range s  — соберётся, но в c окажется номер. Ловушка.

   for _, c := range s   — номер не нужен
   for i := range s      — буква не нужна
   for range s           — нужно только количество шагов

RANGE РАБОТАЕТ НЕ ТОЛЬКО СО СТРОКАМИ:
   строка         -> номер байта, буква
   слайс/массив   -> номер элемента (0,1,2 подряд), элемент
   map            -> ключ, значение
   число (1.22+)  -> 0,1,2... до числа     for i := range 5

ПРИМЕРЫ, ЧАСТЬ 1 (базовая механика)
================================================================================

    package main

    import "fmt"

    func main() {
        // сюда вставляется тело примера
    }

--- 1 --- строка целиком
    s := "Hello"
    fmt.Println(s)
    -> Hello

--- 2 --- отличие от 1: длина вместо строки
    s := "Hello"
    fmt.Println(len(s))
    -> 5

--- 3 --- отличие от 2: строка русская
    s := "Привет"
    fmt.Println(len(s))
    -> 12          БУКВ 6, А БАЙТОВ 12

--- 4 --- отличие от 3: считаем шаги range
    s := "Привет"
    count := 0
    for range s {
        count++
    }
    fmt.Println(count)
    -> 6           range идёт ПО БУКВАМ

--- 5 --- отличие от 4: печатаем номер и букву
    s := "Привет"
    for i, c := range s {
        fmt.Println(i, string(c))
    }
    -> 0 П / 2 р / 4 и / 6 в / 8 е / 10 т
       номера скачут через 2 — это номера БАЙТОВ

--- 6 --- отличие от 5: номер не нужен
    s := "Привет"
    for _, c := range s {
        fmt.Println(string(c))
    }
    -> каждая буква с новой строки

--- 7 --- отличие от 6: считаем нужную букву
    s := "Hello"
    count := 0
    for _, c := range s {
        if c == 'l' {
            count++
        }
    }
    fmt.Println(count)
    -> 2

--- 8 --- отличие от 7: обращение по номеру
    s := "Hello"
    fmt.Println(s[0])
    -> 72          НЕ БУКВА, А ЧИСЛО (байт)

--- 9 --- отличие от 8: просим показать как букву
    s := "Hello"
    fmt.Println(string(s[0]))
    -> H

--- 10 --- отличие от 7: собираем новую строку
    s := "Hello"
    out := ""
    for _, c := range s {
        if c != 'l' {
            out = out + string(c)
        }
    }
    fmt.Println(out)
    -> Heo

    s := "abc"
    for _, c := range s {
        fmt.Println(string(c))
    }
    -> a / b / c  (каждая с новой строки)

--- 6 --- отличие от 5: Print вместо Println
    s := "abc"
    for _, c := range s {
        fmt.Print(string(c))
    }
    fmt.Println()
    -> abc

--- 7 --- отличие от 6: добавили разделитель
    s := "abc"
    for _, c := range s {
        fmt.Print(string(c), "-")
    }
    fmt.Println()
    -> a-b-c-

--- 8 --- отличие от 5: печатаем и номер
    s := "abc"
    for i, c := range s {
        fmt.Println(i, string(c))
    }
    -> 0 a / 1 b / 2 c

--- 9 --- отличие от 8: только номер
    s := "abc"
    for i := range s {
        fmt.Println(i)
    }
    -> 0 / 1 / 2

--- 10 --- отличие от 9: считаем шаги
    s := "abc"
    count := 0
    for range s {
        count++
    }
    fmt.Println(count)              -> 3

--- 11 --- считаем конкретную букву
    s := "banana"
    count := 0
    for _, c := range s {
        if c == 'a' {
            count++
        }
    }
    fmt.Println(count)              -> 3

--- 12 --- отличие от 11: != вместо ==
    s := "banana"
    count := 0
    for _, c := range s {
        if c != 'a' {
            count++
        }
    }
    fmt.Println(count)              -> 3  (b,n,n — совпало случайно!)
    ПРОВЕРЬ НА "hello": c == 'l' даст 2, c != 'l' даст 3

--- 13 --- считаем цифры
    s := "a1b2c3"
    count := 0
    for _, c := range s {
        if c >= '0' && c <= '9' {
            count++
        }
    }
    fmt.Println(count)              -> 3

--- 14 --- отличие от 13: диапазон букв вместо цифр
    s := "a1b2c3"
    count := 0
    for _, c := range s {
        if c >= 'a' && c <= 'z' {
            count++
        }
    }
    fmt.Println(count)              -> 3

--- 15 --- отличие от 14: два диапазона через ||
    s := "Hello World"
    count := 0
    for _, c := range s {
        if (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') {
            count++
        }
    }
    fmt.Println(count)              -> 10
    ЭТО ЗАДАНИЕ С ЭКЗАМЕНА 31.07

--- 16 --- накопитель-строка
    s := "abc"
    out := ""
    for _, c := range s {
        out = out + string(c)
    }
    fmt.Println(out)                -> abc

--- 17 --- отличие от 16: ПОРЯДОК склейки
    s := "abc"
    out := ""
    for _, c := range s {
        out = string(c) + out
    }
    fmt.Println(out)                -> cba

--- 18 --- отличие от 16: с фильтром
    s := "a1b2c3"
    out := ""
    for _, c := range s {
        if c >= 'a' && c <= 'z' {
            out = out + string(c)
        }
    }
    fmt.Println(out)                -> abc

--- 19 --- отличие от 16: каждую букву дважды
    s := "abc"
    out := ""
    for _, c := range s {
        out = out + string(c) + string(c)
    }
    fmt.Println(out)                -> aabbcc

--- 20 --- отличие от 16: сдвиг регистра
    s := "abc"
    out := ""
    for _, c := range s {
        out = out + string(c-32)
    }
    fmt.Println(out)                -> ABC
    (заглавные и строчные отстоят на 32; яснее писать c-'a'+'A')

НАПОМИНАЛКА

  ТРИ МЕСТА НАКОПИТЕЛЯ:
      out := ""            <- создать ДО цикла
      for _, c := range s {
          out = out + ...  <- менять ВНУТРИ
      }
      fmt.Println(out)     <- использовать ПОСЛЕ

  Создашь внутри цикла — на каждом шаге появится новая переменная,
  всё накопленное потеряется. Накопитель обязан жить дольше цикла.

  ПУСТО ДЛЯ СВОЕГО ТИПА:
      число  -> 0
      строка -> ""

  ТРИ ЗНАКА:
      :=   создать и положить      x := 0
      =    положить в готовую      x = x + i
      ==   спросить «равны ли»     x == 7
