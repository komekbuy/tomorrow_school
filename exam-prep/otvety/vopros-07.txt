------------------------------------------------------------------------------
  ЧТО ПРОСЯТ СДЕЛАТЬ
------------------------------------------------------------------------------
  Программа получает ровно два аргумента командной строки: строку-массив
  вида "[1, 2, 3, 4, 5]" и строку с целевой суммой вида "6". Нужно найти все
  пары элементов массива, которые в сумме дают цель, и напечатать ИНДЕКСЫ
  этих элементов в формате "Pairs with sum 6: [[0 4] [1 3]]". Если ни одной
  пары нет — печатается "No pairs found.". Если цель не разбирается как
  целое число — "Invalid target sum.", если элемент массива не число —
  "Invalid number: <элемент>", если формат вообще не тот (нет квадратных
  скобок, не два аргумента) — "Invalid input.". В массиве могут быть
  отрицательные числа.


------------------------------------------------------------------------------
  КАК ДУМАТЬ НАД ЗАДАЧЕЙ (читать до кода)
------------------------------------------------------------------------------
  Дано: две строки из командной строки. Надо: список пар индексов. Задача
  честно делится на три независимых этапа — проверить и разобрать вход,
  найти пары, напечатать. Пиши их по очереди, не смешивая, иначе запутаешься
  в порядке сообщений об ошибках. Порядок проверок диктуется самими
  примерами README, и он единственный, который им не противоречит: сначала
  число аргументов и формат скобок ("Invalid input."), потом цель ("Invalid
  target sum."), потом числа внутри массива ("Invalid number: ...").
  Обоснование: в примере "[1, 2, 3, 4, 20, -4, 5]" "2 5" массив полностью
  корректен, а ответ — "Invalid target sum.", значит цель проверяется
  отдельно; в примере "[1, 2, 3, 4" "5" цель корректна, а ответ — "Invalid
  input.", значит формат скобок проверяется раньше всего. Разбор массива:
  отрезаешь первую и последнюю скобку, режешь остаток по запятой, у каждого
  куска срезаешь пробелы и превращаешь в число через strconv.Atoi; Atoi
  возвращает два значения — число и ошибку, и именно ошибка отличает "5" от
  "p". Поиск пар: массив маленький, поэтому берётся самый прямой способ —
  двойной цикл, внешний i от 0, внутренний j от i+1. Старт j = i+1 (а не j =
  0) сразу даёт два нужных свойства: элемент не сложится сам с собой и пара
  [0 4] не продублируется как [4 0]. Порядок обхода этого цикла совпадает с
  порядком пар в README ([[0 4] [1 3]]), поэтому дополнительно сортировать
  ничего не нужно. Самое скользкое место условия — фраза «each element is
  used only once in a pair, although the element can be repeated in
  different pairs»: она означает, что внутри ОДНОЙ пары индекс встречается
  один раз, но один и тот же индекс МОЖЕТ попасть в несколько разных пар.
  Значит помечать использованные элементы не надо: на [1, 1, 2] с целью 3
  верный ответ [[0 2] [1 2]], индекс 2 повторяется. Печать бесплатная:
  значение типа [][]int Go выводит как [[0 4] [1 3]] — ровно как в README,
  поэтому строку вручную собирать не нужно, хватает fmt.Printf с %v.

  СВЕРЕНО С ЭТАЛОНОМ ШКОЛЫ: 🔴 Поведение в краевых случаях взято из ЭТАЛОНА
  школы, а не из текста задания — они местами расходятся, а бот сверяет
  ответ именно с эталоном. Обрати внимание, что происходит при отсутствии
  аргументов и при пустом результате: молчание и печать пустой строки — это
  разные вещи, и бот их различает.

  ПРО ПАКЕТЫ: здесь приведён код ЭТАЛОНА ШКОЛЫ — ровно тот, с которым бот
  сравнивает твой ответ. Значит все пакеты, которые он подключает, для этого
  задания разрешены: будь иначе, проверка не собралась бы.

    package main
    
    import (
    	"fmt"
    	"os"
    	"strconv"
    	"strings"
    )
    
    func findPairs(arr []int, targetSum int) [][]int {
    	var pairs [][]int
    	for i := 0; i < len(arr); i++ {
    		for j := i + 1; j < len(arr); j++ {
    			if arr[i]+arr[j] == targetSum {
    				pairs = append(pairs, []int{i, j})
    			}
    		}
    	}
    	return pairs
    }
    
    func isValidArrayFormat(s string) bool {
    	s = strings.TrimSpace(s)
    	if len(s) < 2 || s[0] != '[' || s[len(s)-1] != ']' {
    		return false
    	}
    	return true
    }
    
    func main() {
    	if len(os.Args) != 3 {
    		fmt.Println("Invalid input.")
    		return
    	}
    
    	arrayStr := os.Args[1]
    	targetStr := os.Args[2]
    
    	if !isValidArrayFormat(arrayStr) {
    		fmt.Println("Invalid input.")
    		return
    	}
    
    	arrayStr = strings.Trim(arrayStr, "[]")
    	strNums := strings.Split(arrayStr, ",")
    	var arr []int
    	for _, strNum := range strNums {
    		s := strings.TrimSpace(strNum)
    		num, err := strconv.Atoi(s)
    		if err != nil {
    			fmt.Printf("Invalid number: %s\n", s)
    			return
    		}
    		arr = append(arr, num)
    	}
    
    	targetSum, err := strconv.Atoi(targetStr)
    	if err != nil {
    		fmt.Println("Invalid target sum.")
    		return
    	}
    
    	pairs := findPairs(arr, targetSum)
    	if len(pairs) > 0 {
    		fmt.Printf("Pairs with sum %d: %v\n", targetSum, pairs)
    	} else {
    		fmt.Println("No pairs found.")
    	}
    }

------------------------------------------------------------------------------
  ЧТО ПРОСЯТ СДЕЛАТЬ
------------------------------------------------------------------------------
  Программа получает одну строку аргументом и печатает её слова в обратном
  порядке, через один пробел, а в конце — перевод строки. Если аргументов не
  ровно один, программа не печатает вообще ничего, даже перевода строки.
  README обещает, что в проверяемых строках лишних пробелов не будет: ни по
  краям, ни двойных между словами.


------------------------------------------------------------------------------
  КАК ДУМАТЬ НАД ЗАДАЧЕЙ (читать до кода)
------------------------------------------------------------------------------
  Дано: одна строка. Надо: те же слова, но в обратном порядке. Строка в Go —
  это последовательность байтов, а не готовый список слов, поэтому первый
  шаг всегда один и тот же: разрезать строку на слова и сложить их в срез.
  После этого задача становится тривиальной — пройти срез с конца к началу.
  Приём разреза стоит выучить наизусть, он повторяется в трёх заданиях этого
  билета: ведёшь один индекс i по строке и чередуешь два действия —
  «пропусти разделители» и «прочитай слово до следующего разделителя». Перед
  вторым действием запоминаешь позицию начала в переменную start, после него
  вырезаешь подстроку s[start:i] и кладёшь в срез. Такой разрез сам по себе
  схлопывает любое количество подряд идущих пробелов, поэтому программа
  переживёт и «лишние пробелы», которых README обещает не давать — писать
  отдельную обработку не нужно. Сборка ответа: цикл j от len(words)-1 вниз
  до 0 включительно, шаг j--. Пробел добавляется НЕ после каждого слова, а
  только когда j > 0, то есть между словами — иначе в конце строки повиснет
  лишний пробел и побайтовая проверка провалится. Отдельно продумай два
  разных «пустых» случая, их путают чаще всего. Первый: аргументов не один —
  надо не напечатать ничего, ноль байтов, поэтому return стоит до всякой
  печати. Второй: аргумент один, но пустой — слов ноль, результат пустая
  строка, но перевод строки печатается, потому что README показывает `go run
  . "" | cat -e` и одну пустую строку в выводе. Значит \n добавляется
  всегда, кроме случая с неверным числом аргументов. И помни про os.Args[0]:
  это имя самой программы, поэтому «один аргумент» проверяется как
  len(os.Args) == 2.

  ПРО ПАКЕТЫ: здесь приведён код ЭТАЛОНА ШКОЛЫ — ровно тот, с которым бот
  сравнивает твой ответ. Значит все пакеты, которые он подключает, для этого
  задания разрешены: будь иначе, проверка не собралась бы.

    	"fmt"
    	"os"
    	"strings"

    func main() {
    	if len(os.Args) == 2 {
    		a := strings.Split(os.Args[1], " ")
    		for i := len(a) - 1; i >= 0; i-- {
    			fmt.Print(a[i])
    			if i != 0 {
    				fmt.Print(" ")
    			}
    		}
    		fmt.Println()
    	}
    }

  Точка входа.

	if len(os.Args) == 2 {
  os.Args — срез строк (список переменной длины) с аргументами, os.Args[0] — всегда имя самой программы. Длина 2 = ровно один настоящий аргумент.

  Split режет строку по указанному разделителю и возвращает срез строк — куски между разделителями. := создаёт переменную a и кладёт туда результат.
  🔴 Split — НЕ Fields. Split режет ровно по одному указанному символу и пустые куски НЕ выбрасывает: между двумя пробелами подряд появится пустой элемент "". Здесь так можно только потому, что README задания обещает отсутствие лишних пробелов во входных данных.

		for i := len(a) - 1; i >= 0; i-- {
  Обход среза с конца к началу — именно так получается обратный порядок слов, без всякого разворота массива.
  i := len(a) - 1 — старт с последнего индекса; минус один потому, что индексы идут от 0 до len-1;
  i >= 0 — знак >= , а не > : ноль — тоже нормальный индекс, первое слово обязано попасть в вывод;

			fmt.Print(a[i])
  Печатаем слово без перевода строки. Print, а не Println — иначе каждое слово ушло бы на свою строку.

			if i != 0 {
				fmt.Print(" ")
			}

  Это стандартный приём «печатать разделитель для всех, кроме крайнего» — на экзамене он встречается почти в каждом задании с выводом списка.

		}

  * os.Args — срез строк с аргументами командной строки; os.Args[0] это имя
  программы, поэтому один пользовательский аргумент даёт длину 2.
  * срез (slice) — список значений одного типа переменной длины; []string{}
  создаёт пустой срез строк.
  * байт — то, что возвращает обращение к строке по индексу: s[i] имеет тип
  byte, а не «символ».
  * руна (rune) — символ Юникода целиком; здесь не нужна, потому что
  разделители (пробел, табуляция) — однобайтовые, а байты многобайтовых
  символов никогда с ними не совпадают.
  * подстрока s[start:i] — кусок строки от start до i-1; верхняя граница не
  включается.
  * append — добавляет элемент в конец среза и возвращает новый срез,
  результат надо присваивать обратно.
  * len — длина: у строки в байтах, у среза в элементах.
  * += для строк — дописать справа, создав новую строку.
  * os.Stdout.WriteString — запись строки в стандартный вывод без пакета
  fmt.
  * короткое замыкание && — если левая часть условия ложна, правая не
  вычисляется; на этом держится защита от выхода за границу строки.
  * i++ / j-- — увеличить или уменьшить переменную на единицу.
  * for без счётчика — форма «пока условие истинно».

    		}

  Первое, что нужно понять про это задание: это ФУНКЦИЯ, а не программа. Она
  ничего не печатает — она возвращает строку, а печатает уже вызывающий код
  через fmt.Print. Отсюда важное следствие: перевод строки \n обязан быть
  ЧАСТЬЮ возвращаемого значения, иначе в тестовой программе четыре вызова
  fmt.Print склеятся в одну кашу. Дальше разбери требования условия по весу.
  Требование про пустую строку — самое первое и самое буквальное: если str
  == "", возвращаем "Invalid Output" плюс \n (что \n там есть, видно из

  -e). Эту проверку ставь в самое начало и делай её отдельно от проверки
  «слов не нашлось», потому что README показывает третий вызов WordFlip("
  ") и ждёт от него ПУСТУЮ строку вывода, а не Invalid Output. То есть
  строка из пяти пробелов не пустая: она не проходит условие str == "",
  разбирается на ноль слов, и результат получается "" + "\n". Два очень
  похожих входа — два разных ответа, и различает их ровно проверка на точное
  равенство пустой строке. Остальное — тот же разрез на слова, что и в
  revwstr: один индекс i идёт по строке, чередуя «пропусти пробелы и
  табуляции» и «прочитай слово до следующего разделителя», куски
  складываются в срез. Требования «ignore multiple spaces» и «trim
  leading/trailing spaces» при этом выполняются сами собой: цикл,
  пропускающий подряд идущие разделители, просто не создаёт пустых слов,
  поэтому отдельного кода для обрезки писать не надо. Сборка ответа — проход
  по срезу от последнего индекса к нулевому, пробел добавляется только когда
  j > 0, чтобы в конце не осталось лишнего. И последнее: функция не
  импортирует ничего вообще. Для проверки piscine это самый безопасный

  перевод строки словами не разделяют. В конце к собранной строке

    func WordFlip(arg string) string {
    	if arg == "" {
    		return "Invalid Output\n"
    	}
    	var str []string = strings.Split(arg, " ")
    	_len := len(str)
    	var str1 string
    	for i := _len - 1; i >= 0; i-- {
    		if len(str[i]) != 0 {
    			str1 += str[i]
    		}
    		if i > 0 && len(str[i-1]) != 0 {
    			str1 += " "
    		}
    	}
    	return (strings.TrimSpace(str1) + "\n")
    }
