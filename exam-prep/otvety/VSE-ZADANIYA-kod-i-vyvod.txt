      import "strings"

      	if len(str) == 0 {
      		return ""
      	} else if len(str) == 1 {
      		return (str)
      	} else {
      		var res strings.Builder
      		i := 0
      		for i = 0; i < int(len(str)/2); i++ {
      			res.WriteRune(rune(str[i]))
      		}
      		return res.String()
      	}
      }

  CAMELTOSNAKECASE  ..................................... [КОД ЭТАЛОНА ШКОЛЫ]
  тип: функция
  прототип: func CamelToSnakeCase(s string) string

  ЗАДАНИЕ (своими словами):
      Функция получает строку и должна вернуть её в записи snake_case: перед
      каждой заглавной буквой ставится знак подчёркивания. Если строка пустая
      — вернуть пустую строку. Если строка не является правильным camelCase —
      вернуть её без изменений. Правильным camelCase считается строка, которая
      состоит только из букв (без цифр и знаков), не заканчивается заглавной
      буквой и не содержит двух заглавных букв подряд. Буквы при переводе НЕ
      меняют регистр: "HelloWorld" превращается в "Hello_World".

  ТЕКСТ ЗАДАНИЯ (дословно из школы):
      Write a function that converts a `string` from `camelCase` to
      `snake_case`.

      - If the `string` is empty, return an empty `string`.
      - If the `string` is not `camelCase`, return the `string` unchanged.
      - If the `string` is `camelCase`, return the `snake_case` version of the
      `string`.

      For this exercise you need to know that `camelCase` has two different
      writing alternatives that will be accepted:

      - lowerCamelCase
      - UpperCamelCase

      Rules for writing in `camelCase`:

      - The word does not end on a capitalized letter (CamelCasE).
      - No two capitalized letters shall follow directly each other
      (CamelCAse).
      - Numbers or punctuation are not allowed in the word anywhere
      (camelCase1).

  ОЖИДАЕМЫЙ ВЫВОД:
      $ go run .
      Hello_World
      hello_World
      camel_Case
      CAMELtoSnackCASE
      camel_To_Snake_Case
      hey2

  КОД:
      package piscine
      
      func containOnlyAlphabet(str string) bool {
      	for _, c := range str {
      		if (c < 'a' || c > 'z') && (c < 'A' || c > 'Z') {
      			return false
      		}
      	}
      	return true
      }
      
      func isUpper(s rune) bool {
      	return s >= 'A' && s <= 'Z'
      }
      
      func CamelToSnakeCase(s string) string {
      	result := ""
      	if len(s) == 0 || !containOnlyAlphabet(s) {
      		return s
      	}
      	for i := 0; i < len(s); i++ {
      		if i != 0 && isUpper(rune(s[i])) && i+1 < len(s) && !isUpper(rune(s[i+1])) {
      			result += "_"
      			result += string(s[i])
      		} else if !isUpper(rune(s[i])) || (i == 0 && isUpper(rune(s[i]))) {
      			result += string(s[i])
      		} else {
      			return s
      		}
      	}
      	return result
      }


  COUNTREPEATS  ......................................... [КОД ЭТАЛОНА ШКОЛЫ]
  тип: функция

  ЗАДАНИЕ (своими словами):
      Функция получает строку и сжимает в ней подряд идущие одинаковые
      символы. Каждая цепочка одинаковых символов заменяется на сам символ и
      число повторов. Если символ встретился подряд только один раз, число не

      подряд идущих повторов.

  ТЕКСТ ЗАДАНИЯ: в публичной репе школы отсутствует

  ОЖИДАЕМЫЙ ВЫВОД: примера в задании нет

  КОД:
      package piscine
      
      import (
      	"strconv"
      )
      
      func CountRepeats(s string) string {
      	if s == "" {
      		return s
      	}
      
      	var result string
      	var counter = 1
      
      	for i := 0; i < len(s); i++ {
      		if i+1 < len(s) && string(s[i]) == string(s[i+1]) {
      			counter++
      		} else {
      			result += string(s[i])
      			if counter > 1 {
      				result += strconv.Itoa(counter)
      			}
      			counter = 1
      		}
      	}
      	return result
      }

  ТЕКСТ ЗАДАНИЯ (дословно из школы):

      - The second `int` must be between **_2_** and **_36_**. If not, the
      function returns `-1`.
      - If the first `int` is negative, reverse the sign and count the digits.

  ОЖИДАЕМЫЙ ВЫВОД:

      package piscine
      
      func DigitLen(n, base int) int {
      
      	if base < 2 || base > 36 {
      		return -1
      	}
      
      	if n < 0 {

      func WordAnatomy(initialWord string, prefixArray []string, suffixArray []string) string {
      	matchedPrefix := ""
      	matchedSuffix := ""
      
      	for _, prefix := range prefixArray {
      		if hasPrefix(initialWord, prefix) {
      			matchedPrefix = prefix
      			break
      		}
      	}
      	for _, suffix := range suffixArray {
      		if hasSuffix(initialWord, suffix) {
      			matchedSuffix = suffix
      			break
      		}
      	}

      		return false
      	}

      }

      	if len(suffix) > len(initialWord) {
      		return false
      	}
      
      	return initialWord[len(initialWord)-len(suffix):] == suffix
      }

      import (
      	"math"
      	"unicode"
      )
      
      func Entropy(password string) float64 {
      	const (
      		lowerBit = 1 << iota
      		upperBit
      		digitBit
      		specialBit
      	)
      
      	used := 0
      	poolSize := 0.0
      
      	for _, r := range password {
      		if used&lowerBit == 0 && unicode.IsLower(r) {
      			used |= lowerBit
      			poolSize += 26
      		} else if used&upperBit == 0 && unicode.IsUpper(r) {
      			used |= upperBit
      			poolSize += 26
      		} else if used&digitBit == 0 && unicode.IsDigit(r) {
      			used |= digitBit
      			poolSize += 10
      		} else if used&specialBit == 0 && !unicode.IsLetter(r) && !unicode.IsDigit(r) {
      			used |= specialBit
      			poolSize += 32
      		}
      
      		if used == lowerBit|upperBit|digitBit|specialBit {
      			break
      		}
      	}
      
      	if poolSize == 0 {
      		return 0
      	}
      
      	return float64(len(password)) * math.Log2(poolSize)
      }

      import "os"
      
      func atoi(s string) (int, bool) {
      	if len(s) == 0 {
      		return 0, false
      	}
      	n := 0
      	for i := 0; i < len(s); i++ {
      		if s[i] < '0' || s[i] > '9' {
      			return 0, false
      		}
      		n = n*10 + int(s[i]-'0')
      	}
      	return n, true
      }
      
      func itoa(n int) string {
      	if n == 0 {
      		return "0"
      	}
      	result := ""
      	for n > 0 {
      		result = string(rune('0'+n%10)) + result
      		n /= 10
      	}
      	return result
      }
      
      func isPrime(n int) bool {
      	if n < 2 {
      		return false
      	}
      	for i := 2; i*i <= n; i++ {
      		if n%i == 0 {
      			return false
      		}
      	}
      	return true
      }
      
      func main() {
      	if len(os.Args) != 2 {
      		os.Stdout.WriteString("0\n")
      		return
      	}
      	n, ok := atoi(os.Args[1])
      	if !ok || n <= 0 {
      		os.Stdout.WriteString("0\n")
      		return
      	}
      	sum := 0
      	for i := 2; i <= n; i++ {
      		if isPrime(i) {
      			sum += i
      		}
      	}
      	os.Stdout.WriteString(itoa(sum) + "\n")
      }

      import "strings"
      
      func FifthAndSkip(str string) string {
      	if str == "" {
      		return "\n"
      	}

      		return "Invalid Input\n"
      	}
      	s := strings.ReplaceAll(str, " ", "")

      		if j == 5 {

      			j++
      		}

      }

      func RevConcatAlternate(slice1, slice2 []int) []int {
      	len1 := len(slice1)
      	len2 := len(slice2)
      	maxLen := len1
      	result := make([]int, 0, len1+len2)
      
      	if len2 > maxLen {
      		maxLen = len2
      	}

      		if i < len1 {

      		}
      		if i < len2 {
      			result = append(result, slice2[i])
      		}
      	}
      
      	return result
      }

      func ifNegative(a []string, n int) int {
      	if n < 0 {
      		n = len(a) + n
      	}
      
      	if n < 0 {
      		n = 0
      	} else if n > len(a) {
      		n = len(a)
      	}

      }
      
      func Slice(a []string, nbr ...int) []string {
      	if len(nbr) == 0 {
      		return a
      	}
      
      	first := nbr[0]
      	if len(nbr) == 1 {
      		if first < 0 {
      			first = len(a) + first
      			if first < 0 {
      				return a
      			}
      		}
      		return a[first:]
      	}
      	second := nbr[1]
      
      	first = ifNegative(a, first)
      	second = ifNegative(a, second)
      
      	if first > second {
      		return nil
      	}
      
      	return a[first:second]
      }

      			result += v.romanDigit
      			nbr -= v.num

      		}
      	}
      	a = strings.Join(result2, "+")
      	return a
      }

      package main
      
      import (
      	"fmt"
      	"os"
      )
      
      func matchBrackets(exp string) bool {
      	runes := []rune(exp)
      	var opened []rune
      	ptr := -1
      	for _, c := range runes {
      		if c == '(' || c == '[' || c == '{' {
      			opened = append(opened, c)
      			ptr++
      		} else if c == ')' {
      			if ptr < 0 || opened[ptr] != '(' {
      				return false
      			}
      			opened = opened[:len(opened)-1]
      			ptr--
      		} else if c == ']' {
      			if ptr < 0 || opened[ptr] != '[' {
      				return false
      			}
      			opened = opened[:len(opened)-1]
      			ptr--
      		} else if c == '}' {
      			if ptr < 0 || opened[ptr] != '{' {
      				return false
      			}
      			opened = opened[:len(opened)-1]
      			ptr--
      		}
      	}
      	return len(opened) == 0
      }
      
      func main() {
      	if len(os.Args) == 1 {
      		fmt.Println()
      	} else {
      		for _, v := range os.Args[1:] {
      			if matchBrackets(v) {
      				fmt.Println("OK")
      			} else {
      				fmt.Println("Error")
      			}
      		}
      	}
      }

      ```
      1 2 * 3 * 4 -
      2 3 * 4 -
      6 4 -
      ```

      package main
      
      import (
      	"fmt"
      	"os"
      	"strings"
      	"unicode"
      )
      
      func singleSearch(exp []string, text string) []string {
      	items := strings.Split(text, " ")
      	var result []string
      
      	for _, item := range items {
      		for _, word := range exp {
      			if strings.Contains(item, word) {
      				result = append(result, item)
      			}
      		}
      	}
      	return result
      }
      
      func simpleSearch(runes []rune, text string) []string {
      	exp := string(runes)
      
      	var result []string
      	if !strings.ContainsRune(exp, '|') {
      		helper := []string{exp}
      		result = append(singleSearch(helper, text))
      	} else {
      		expWords := strings.Split(exp, "|")
      		result = append(result, singleSearch(expWords, text)...)
      	}
      	return result
      }
      
      func brackets(regexp, text string) {
      	if text == "" || regexp == "" {
      		return
      	}
