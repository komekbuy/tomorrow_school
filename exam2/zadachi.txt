B1. func CountLetters(s string) int
    Сколько латинских букв (пробелы, цифры, знаки не считать).
    "Hello world"  -> 10
    "H e l l o"    -> 5
    "w|o|r|l|d"    -> 5
    ЭТО ЗАДАЧА С ЭКЗАМЕНА 1.

B2. func CountDigits(s string) int
    Сколько цифр в строке.
    "a1b2c3" -> 3      "abc" -> 0

B3. func CountVowels(s string) int
    Сколько гласных a e i o u (обоих регистров).
    "Hello World" -> 3      "xyz" -> 0

B4. func CountUpper(s string) int
    Сколько заглавных латинских букв.
    "Hello World" -> 2

B5. func CountRunes(s string) int
    Сколько БУКВ в строке (не байтов!).
    "Hello" -> 5      "Привет" -> 6      "" -> 0
    ЛОВУШКА: len(s) даст 12 для "Привет". Нужен range.

B6. func CountWords(s string) int
    Сколько слов (разделители — пробелы).
    "Hello how are you" -> 4      "  a  b  " -> 2      "" -> 0

B7. func CountChar(s string, target rune) int
    Сколько раз встречается заданная буква.
    CountChar("banana", 'a') -> 3

C1. func Reverse(s string) string
    Строка задом наперёд.
    "Hello" -> olleH      "abc" -> cba      "" -> ""

C2. func ToUpper(s string) string
    Все строчные латинские -> заглавные, остальное не трогать.
    "Hello! 123" -> HELLO! 123

C3. func ToLower(s string) string
    Обратное к C2.
    "Hello! 123" -> hello! 123

C4. func RemoveSpaces(s string) string
    Убрать все пробелы.
    "a b c" -> abc

C5. func RemoveDigits(s string) string
    Убрать все цифры.
    "a1b2c3" -> abc

C6. func Capitalize(s string) string
    Первая буква каждого слова заглавная, остальные строчные.
    "hello WORLD" -> Hello World

C7. func Repeat(s string, n int) string
    Повторить строку n раз.
    Repeat("ab", 3) -> ababab      Repeat("x", 0) -> ""

ТИПОВЫЕ ОШИБКИ — ПРОВЕРЯТЬ ПЕРЕД СДАЧЕЙ КАЖДОЙ ЗАДАЧИ
================================================================================
Это твои личные грабли, собранные за неделю. Пройтись по списку = поймать 80% провалов.

□ 1. RETURN ПРЕРЫВАЕТ ФУНКЦИЮ.
     return внутри цикла -> выход на первом шаге. Обычно return нужен ПОСЛЕ цикла.

□ 2. НАКОПИТЕЛЬ СОЗДАЁТСЯ ДО ЦИКЛА.
     Внутри цикла := создаст новую переменную на каждом шаге, всё потеряется.
     число -> 0, строка -> ""

□ 3. ТРИ ЗНАКА.
     :=  создать и положить     =  положить в готовую     ==  спросить «равны ли»
     return + присваивание в одной строке НЕ РАБОТАЮТ.

□ 4. ГРАНИЦЫ ЦИКЛА.
     Выписать первый и последний шаг ДО запуска.
     Сумма 1..n требует i := 1; i <= n, а не i := 0; i < n.

□ 5. ДИАПАЗОН НЕ ПЕРЕВЁРНУТ.
     Правильно: c >= 'a' && c <= 'z'
     Подставь букву и прочитай вслух: «m больше a? да. m меньше z? да».

□ 6. && ВНУТРИ ДИАПАЗОНА, || МЕЖДУ ДИАПАЗОНАМИ.
     (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
     Скобки обязательны.

□ 7. РУНА ПРОТИВ ЧИСЛА.
     '0' это символ (код 48), 0 это число. Сравнивать символы с символами.

□ 8. s[i] ДАЁТ БАЙТ, range ДАЁТ БУКВУ.
     На кириллице s[i] сломается. Считаешь буквы — только range.

□ 9. gofmt ПЕРЕД СДАЧЕЙ. Всегда.

□ 10. ПЕРЕЧИТАТЬ УСЛОВИЕ ПОСЛЕ НАПИСАНИЯ КОДА.
      Имя функции может врать (isalpha требовал БУКВЫ И ЦИФРЫ).
      Пример в задании — последняя инстанция.
