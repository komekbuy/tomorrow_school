func CountDigits(s string) int {
	r := []rune(s)
	count := 0
	for _, ch := range r {
		if ch >= '0' && ch <= '9' {
			count++
		}
	}
	return count
}

// Считает цифры. От эталона отличается одной строкой: вместо двух диапазонов
// букв — один диапазон цифр '0'..'9'. Кавычки обязательны: '0' это символ ноль
// (код 48), а 0 без кавычек — само число ноль.

fmt.Println(CountDigits("a1b2c3"))   -> 3
fmt.Println(CountDigits("abc"))      -> 0

------------------------------------------------------------ 1.2
func CountUpper(s string) int {
	r := []rune(s)
	count := 0
	for _, ch := range r {
		if ch >= 'A' && ch <= 'Z' {
			count++
		}
	}
	return count
}

// Считает только ЗАГЛАВНЫЕ буквы. Половина условия эталона — второй диапазон

fmt.Println(CountUpper("Hello World"))   -> 2
fmt.Println(CountUpper("abc"))           -> 0
fmt.Println(CountUpper("ABC"))           -> 3

		}
	}

func CountSpaces(s string) int {
	r := []rune(s)
	count := 0
	for _, ch := range r {
		if ch == ' ' {
			count++
		}
	}
	return count
}

// Считает пробелы. Диапазон сменился на точное сравнение ==, потому что

fmt.Println(CountSpaces("a b c"))   -> 2
fmt.Println(CountSpaces("abc"))     -> 0
fmt.Println(CountSpaces("   "))     -> 3

------------------------------------------------------------ 1.5
func CountVowels(s string) int {
	r := []rune(s)
	count := 0
	for _, ch := range r {
		if ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u' {
			count++
		}
	}
	return count
}

// Считает гласные. Диапазоном их не задать — буквы разбросаны по алфавиту,
// поэтому пять отдельных сравнений через ||. Каждое сравнение полное:
// ch == 'a' || ch == 'e', а не ch == 'a' || 'e'.

fmt.Println(CountVowels("hello world"))   -> 3
fmt.Println(CountVowels("xyz"))           -> 0
fmt.Println(CountVowels("aeiou"))         -> 5

ЧЕМ ОТЛИЧАЕТСЯ ОТ БЛОКА 1: искомое не вшито в код, а приходит параметром.

------------------------------------------------------------ 2.1
func CountNotChar(str string, c rune) int {
	s := []rune(str)
	count := 0
	for _, ch := range s {
		if ch != c {
			count++
		}
	}
	return count
}

// Считает символы, которые НЕ равны заданному. Один знак отличия: != вместо ==.
// Проверка на понимание: CountChar + CountNotChar всегда дают длину строки.

fmt.Println(CountNotChar("hello", 'l'))   -> 3
fmt.Println(CountNotChar("aaa", 'a'))     -> 0
fmt.Println(CountNotChar("", 'x'))        -> 0

------------------------------------------------------------ 2.2
func CountTwoChars(str string, a, b rune) int {
	s := []rune(str)
	count := 0
	for _, ch := range s {
		if ch == a || ch == b {
			count++
		}
	}
	return count
}

// Считает вхождения ДВУХ символов сразу. Параметров стало три; a, b rune —
// это короткая запись для a rune, b rune. Пишется только когда тип общий.

fmt.Println(CountTwoChars("hello world", 'l', 'o'))   -> 5
fmt.Println(CountTwoChars("abc", 'x', 'y'))           -> 0
fmt.Println(CountTwoChars("aabb", 'a', 'b'))          -> 4

------------------------------------------------------------ 2.3
func CountGreaterThan(str string, c rune) int {

	count := 0

// Считает символы, у которых КОД больше заданного. Работает, потому что руна —

fmt.Println(CountGreaterThan("abc", 'a'))   -> 2
fmt.Println(CountGreaterThan("123", '5'))   -> 0

// Зеркало предыдущего. Смотри на "123" с границей '5': ответ 3, потому что

fmt.Println(CountLessThan("123", '5'))   -> 3

func CountInRange(str string, lo, hi rune) int {

	count := 0
	for _, ch := range s {
		if ch >= lo && ch <= hi {

		}

func PrintIf(str string) string {
	s := []rune(str)
	if len(s) >= 3 {
		return "G\n"
	}

}

ГЛАВНОЕ В БЛОКЕ: цикла нет вообще. Функция ВОЗВРАЩАЕТ строку, а не печатает.
	- "\n" внутри возвращаемой строки, поэтому зовут через fmt.Print, не Println
	- второй return БЕЗ else — до него доходит только тот случай, где if не сработал
	- это ровно та ошибка, что стоила тебе двух часов: печать вместо возврата

func SayLong(str string) string {
	s := []rune(str)
	if len(s) >= 5 {
		return "Long\n"
	}

	return "Short\n"
}

// Длинное слово или короткое. Отличие от эталона — только число в условии
// и текст ответов. Скелет тот же до знака.

fmt.Print(SayLong("hello"))   -> Long
fmt.Print(SayLong("hi"))      -> Short
fmt.Print(SayLong(""))        -> Short

func CheckEmpty(str string) string {

	if len(s) == 0 {
		return "Empty\n"
	}

	return "Not empty\n"
}

// Пустая строка или нет. Сравнение стало точным: == 0. Строка из трёх пробелов

fmt.Print(CheckEmpty(""))      -> Empty

func SizeWord(str string) string {
	s := []rune(str)
	if len(s) <= 2 {
		return "Small\n"
	}
	if len(s) <= 5 {
		return "Medium\n"
	}

	return "Big\n"
}

// Три ответа вместо двух. Второй if НЕ проверяет нижнюю границу — если бы длина

fmt.Print(SizeWord("go"))         -> Small

func StartsUpper(str string) string {
	s := []rune(str)
	if len(s) == 0 {
		return "Invalid Input\n"
	}
	if s[0] >= 'A' && s[0] <= 'Z' {
		return "Upper\n"
	}

	return "Lower\n"
}

// С заглавной начинается слово или нет. Первый if здесь ОБЯЗАТЕЛЕН: без него

// Проверка длины всегда идёт ПЕРЕД обращением по индексу.

fmt.Print(StartsUpper("Hello"))   -> Upper
fmt.Print(StartsUpper("hello"))   -> Lower
fmt.Print(StartsUpper(""))        -> Invalid Input

func LenParity(str string) string {
	s := []rune(str)
	if len(s)%2 == 0 {
		return "Even\n"
	}

	return "Odd\n"
}

БЛОК 4 — ЧИСЛА + ЗАЩИТА ОТ ПЛОХОГО ВВОДА   эталон: RectPerimeter
################################################################################

ЭТАЛОН (было на экзамене):

func RectPerimeter(w, h int) int {
	if w < 0 && h < 0 {
		return -1
	}

	p := (w + h) * 2
	return p
}

СКЕЛЕТ:
	1. сначала ОТСЕЧКА плохого ввода -> return -1
	2. потом формула
	3. return результата
Отсечка всегда ПЕРВОЙ. Если посчитать сначала, а проверить потом — на плохих
данных успеет посчитаться мусор.

ВНИМАНИЕ на && против ||:
	w < 0 && h < 0   -> -1 только если ОБА отрицательные
	w < 0 || h < 0   -> -1 если ХОТЯ БЫ ОДИН
В эталоне стоит && — читай условие задачи, там написано, какое нужно.

------------------------------------------------------------ 4.1
func RectArea(w, h int) int {
	if w <= 0 || h <= 0 {
		return -1
	}

	a := w * h
	return a
}

// Площадь прямоугольника. Два отличия от эталона: || вместо && (плохо, если
// хоть одна сторона негодная) и <= вместо < (нулевая сторона тоже не сторона).

fmt.Println(RectArea(10, 2))   -> 20
fmt.Println(RectArea(5, 5))    -> 25
fmt.Println(RectArea(10, -2))  -> -1

------------------------------------------------------------ 4.2
func TrianglePerimeter(a, b, c int) int {
	if a < 0 || b < 0 || c < 0 {
		return -1
	}

	p := a + b + c
	return p
}

// Периметр треугольника. Параметров три, значит и проверок три — через ||.
// Каждое сравнение полное: a < 0 || b < 0, а не a || b < 0.

fmt.Println(TrianglePerimeter(3, 4, 5))      -> 12
fmt.Println(TrianglePerimeter(10, 10, 10))   -> 30
fmt.Println(TrianglePerimeter(3, -4, 5))     -> -1

------------------------------------------------------------ 4.3
func SquarePerimeter(a int) int {
	if a < 0 {
		return -1
	}

	p := a * 4
	return p
}

// Периметр квадрата. Параметр один — и проверка одна, без && и ||.
// Ноль здесь пропускается: 0 не меньше 0, поэтому ответ 0, а не -1.

fmt.Println(SquarePerimeter(5))    -> 20
fmt.Println(SquarePerimeter(0))    -> 0
fmt.Println(SquarePerimeter(-3))   -> -1

------------------------------------------------------------ 4.4
func BoxVolume(a, b, c int) int {
	if a <= 0 || b <= 0 || c <= 0 {
		return -1
	}

	v := a * b * c
	return v
}

// Объём коробки. Три параметра и <= — потому что коробка с нулевой стороной

fmt.Println(BoxVolume(2, 3, 4))   -> 24

func Average(a, b int) int {
	if a < 0 && b < 0 {
		return -1
	}

	m := (a + b) / 2
	return m

################################################################################
БЛОК 5 — ЦИКЛ + СРЕЗ ДО/ПОСЛЕ НАЙДЕННОГО   эталон: FirstWorld
################################################################################

ЭТАЛОН (было на экзамене):

func FirstWorld(s string) string {
	str := []rune(s)
	for i, ch := range str {
		if ch == ' ' {
			return string(str[:i]) + "\n"
		}
	}
	return string(str) + "\n"
}

САМЫЙ ВАЖНЫЙ БЛОК — здесь три вещи сразу:
	1. НУЖЕН НОМЕР. Поэтому for i, ch, а не for _, ch — по номеру режем.
	2. return ВНУТРИ цикла = выход на первом же совпадении. Дальше не идём.
	3. return ПОСЛЕ цикла = случай «не нашли ничего». Без него не соберётся.

СРЕЗЫ:
	str[:i]    от начала ДО i, сам i НЕ входит
	str[i+1:]  от следующего за i и до конца
	string(...) обязателен: срез рун — это не строка, печатать его нельзя

------------------------------------------------------------ 5.1
func UpToDigit(s string) string {
	str := []rune(s)
	for i, ch := range str {
		if ch >= '0' && ch <= '9' {
			return string(str[:i]) + "\n"
		}
	}
	return string(str) + "\n"
}

// Кусок строки до первой цифры. От эталона отличается одним условием: вместо
// пробела — диапазон цифр. Если цифр нет — вернётся вся строка.
// Если цифра первая — вернётся пустая строка, и на экране будет пустая строчка.

fmt.Print(UpToDigit("abc123"))   -> abc
fmt.Print(UpToDigit("abc"))      -> abc
fmt.Print(UpToDigit("1abc"))     -> (пустая строка)

------------------------------------------------------------ 5.2
func UpToChar(s string, c rune) string {
	str := []rune(s)
	for i, ch := range str {
		if ch == c {
			return string(str[:i]) + "\n"
		}
	}
	return string(str) + "\n"
}

// Кусок до заданного символа. Здесь сходятся блок 5 и блок 2: скелет со срезом,
// а искомый символ приходит параметром rune.

fmt.Print(UpToChar("hello-world", '-'))   -> hello
fmt.Print(UpToChar("hello", '-'))         -> hello
fmt.Print(UpToChar("-abc", '-'))          -> (пустая строка)

------------------------------------------------------------ 5.3
func UpToDot(s string) string {
	str := []rune(s)
	for i, ch := range str {
		if ch == '.' {
			return string(str[:i]) + "\n"
		}
	}
	return string(str) + "\n"
}

// Имя файла без расширения — кусок до точки. Тот же эталон, другой символ.

fmt.Print(UpToDot("file.txt"))   -> file
fmt.Print(UpToDot("noext"))      -> noext
fmt.Print(UpToDot(".hidden"))    -> (пустая строка)

------------------------------------------------------------ 5.4
func AfterFirstSpace(s string) string {
	str := []rune(s)
	for i, ch := range str {
		if ch == ' ' {
			return string(str[i+1:]) + "\n"
		}
	}
	return "\n"
}

// Всё, что ПОСЛЕ первого пробела. Срез перевернулся: str[i+1:] вместо str[:i].
// i+1 потому, что сам пробел брать не надо. И последний return другой: если
// пробела нет, то и «после пробела» ничего нет — возвращаем пустую строку.

fmt.Print(AfterFirstSpace("hello there"))   -> there
fmt.Print(AfterFirstSpace("hello"))         -> (пустая строка)
fmt.Print(AfterFirstSpace("a b c"))         -> b c

------------------------------------------------------------ 5.5
func LastWord(s string) string {
	str := []rune(s)
	for i := len(str) - 1; i >= 0; i-- {
		if str[i] == ' ' {
			return string(str[i+1:]) + "\n"
		}
	}
	return string(str) + "\n"
}

// Последнее слово. Единственный пример блока, где цикл идёт НАЗАД, поэтому
// range не годится — он умеет только вперёд. Заголовок обычного цикла:
//   i := len(str) - 1   старт с последнего (счёт с нуля, поэтому -1)
//   i >= 0              идём до нулевого включительно
//   i--                 шаг назад
// И раз номер берётся руками, буква достаётся через str[i], а не из range.

fmt.Print(LastWord("hello big world"))   -> world
fmt.Print(LastWord("hello"))             -> hello
fmt.Print(LastWord("a b"))               -> b

БЛОК 6 — ОТСЕЧКА КОРОТКИХ + СРЕЗ       эталон: RetainFirstHalf

ЭТАЛОН (было на экзамене):

func RetainFirstHalf(str string) string {
	if len(str) == 0 {
		return ""
	}
	if len(str) == 1 {
		return str
	}
	s := []rune(str)
	return string(s[:len(str)/2])
}

СКЕЛЕТ: две отсечки, потом срез. Цикла нет.
	пустая строка -> ""
	один символ   -> сам символ (делить нечего)
	дальше        -> режем

ОТЛИЧИЕ ОТ БЛОКА 3: там возвращали ГОТОВЫЙ текст ("G\n"), здесь — КУСОК
исходной строки. И "\n" не добавляется, поэтому зовут через Println.

ПРИМЕЧАНИЕ. В моих примерах длина берётся с рун: len(s), а не len(str).
На английском разницы нет, на кириллице есть: len("Привет") = 12 байтов,
а рун 6. Считать надо то, что режешь.

------------------------------------------------------------ 6.1
func RetainSecondHalf(str string) string {
	if len(str) == 0 {
		return ""
	}
	if len(str) == 1 {
		return str
	}
	s := []rune(str)
	return string(s[len(s)/2:])
}

// Вторая половина строки. Двоеточие переехало: s[len(s)/2:] — от середины и до
// конца. У "Hello" пять букв, середина 5/2 = 2 (деление целых), значит режем
// с третьей буквы: llo.

fmt.Println(RetainSecondHalf("Hello"))   -> llo
fmt.Println(RetainSecondHalf("abcd"))    -> cd
fmt.Println(RetainSecondHalf("a"))       -> a

------------------------------------------------------------ 6.2
func FirstThird(str string) string {
	if len(str) == 0 {
		return ""
	}
	if len(str) == 1 {
		return str
	}
	s := []rune(str)
	return string(s[:len(s)/3])
}

// Первая треть. Делитель сменился с 2 на 3. У "ab" получается 2/3 = 0, и срез
// s[:0] даёт пустую строку — это не поломка, а честный ответ: трети в двух
// буквах нет.

fmt.Println(FirstThird("abcdef"))   -> ab
fmt.Println(FirstThird("abc"))      -> a
fmt.Println(FirstThird("ab"))       -> (пустая строка)

------------------------------------------------------------ 6.3
func DropFirst(str string) string {
	if len(str) == 0 {
		return ""
	}
	if len(str) == 1 {
		return ""
	}
	s := []rune(str)
	return string(s[1:])
}

// Строка без первого символа. Деления нет вообще — число в срезе стоит прямо.
// Вторая отсечка изменилась: из одного символа убрать первый = пустая строка,
// поэтому "" а не str.

fmt.Println(DropFirst("hello"))   -> ello
fmt.Println(DropFirst("ab"))      -> b
fmt.Println(DropFirst("a"))       -> (пустая строка)

------------------------------------------------------------ 6.4
func DropLast(str string) string {
	if len(str) == 0 {
		return ""
	}
	if len(str) == 1 {
		return ""
	}
	s := []rune(str)
	return string(s[:len(s)-1])
}

// Строка без последнего символа. Зеркало предыдущего: s[:len(s)-1] — от начала
// и до последнего, сам последний не входит.

fmt.Println(DropLast("hello"))   -> hell
fmt.Println(DropLast("ab"))      -> a
fmt.Println(DropLast("a"))       -> (пустая строка)

------------------------------------------------------------ 6.5
func MiddleChar(str string) string {
	if len(str) == 0 {
		return ""
	}
	if len(str) == 1 {
		return str
	}
	s := []rune(str)
	return string(s[len(s)/2])
}

// Средний символ. Двоеточия НЕТ вообще — это не срез, а обращение по индексу:
// берём одну руну. Поэтому string(...) здесь превращает не срез, а одно число
// в строку. Без него напечатается код символа.

fmt.Println(MiddleChar("abc"))    -> b
fmt.Println(MiddleChar("abcd"))   -> c
fmt.Println(MiddleChar("a"))      -> a

------------------------------------------------------------ 7.1
func HasUpper(s string) bool {
	str := []rune(s)
	for _, ch := range str {
		if ch >= 'A' && ch <= 'Z' {
			return true
		}
	}
	return false
}

// Есть ли в строке хоть одна заглавная. Условие взято из 1.2, а скелет — из
// эталона блока. Так и собираются задачи: скелет одного блока + условие другого.

fmt.Println(HasUpper("hello"))   -> false
fmt.Println(HasUpper("helLo"))   -> true
fmt.Println(HasUpper(""))        -> false

------------------------------------------------------------ 7.2
func HasSpace(s string) bool {
	str := []rune(s)
	for _, ch := range str {
		if ch == ' ' {
			return true
		}
	}
	return false
}

// Есть ли пробел. Точное сравнение вместо диапазона.
// Строка из одного пробела даёт true — пробел это символ.

fmt.Println(HasSpace("hello world"))   -> true
fmt.Println(HasSpace("hello"))         -> false
fmt.Println(HasSpace(" "))             -> true

------------------------------------------------------------ 7.3
func HasLetter(s string) bool {
	str := []rune(s)
	for _, ch := range str {
		if ch >= 'a' && ch <= 'z' || ch >= 'A' && ch <= 'Z' {
			return true
		}
	}
	return false
}

// Есть ли хоть одна буква. Условие целиком из эталона блока 1 (CountAlpha),
// но ответ уже не число, а да/нет.

fmt.Println(HasLetter("123a"))   -> true
fmt.Println(HasLetter("123"))    -> false
fmt.Println(HasLetter(""))       -> false

------------------------------------------------------------ 7.4
func AllDigits(s string) bool {
	str := []rune(s)
	for _, ch := range str {
		if ch < '0' || ch > '9' {
			return false
		}
	}
	return true
}

// ВСЕ ли символы — цифры. Здесь всё перевёрнуто: ищем НАРУШИТЕЛЯ и на нём
// возвращаем false, а true стоит в конце. Правило: «хоть один» -> true внутри,
// «все» -> false внутри. Условие тоже перевёрнуто: не «в диапазоне», а
// «вне диапазона» — знаки развернулись и && стало ||.
// Пустая строка даёт true: нарушителей нет, значит все цифры.

fmt.Println(AllDigits("2026"))   -> true
fmt.Println(AllDigits("20a6"))   -> false
fmt.Println(AllDigits(""))       -> true

------------------------------------------------------------ 7.5
func IsAllSpaces(s string) bool {
	str := []rune(s)
	for _, ch := range str {
		if ch != ' ' {
			return false
		}
	}
	return true
}

// Строка целиком из пробелов? Тот же перевёрнутый скелет, что и в 7.4, но
// условие проще: != вместо диапазона.

fmt.Println(IsAllSpaces("   "))   -> true
fmt.Println(IsAllSpaces(" a "))   -> false
fmt.Println(IsAllSpaces(""))      -> true
