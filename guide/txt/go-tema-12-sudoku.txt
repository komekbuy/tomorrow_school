1. Набирай руками. Не копируй мышкой — пока печатаешь, глаз замечает
   скобки, точки и кавычки, которые при копировании проходят мимо.

2. ПЕРЕД запуском запиши, что программа выведет. Потом запусти и сверь
   с ответом под примером.

3. Не совпало — это и есть самое ценное место. Разберись, почему вышло
   именно так, и только потом иди дальше.

4. Заработало — сломай нарочно: убери кавычки, поменяй тип, поставь другой
   индекс. Прочитай, что скажет компилятор.

5. Каждый следующий пример отличается от предыдущего одним изменением —
   оно указано в заголовке. Найди его глазами до запуска.

ПРИМЕР 1.  Изменилось: фигуры кончились, началась доска
------------------------------------------------------------------------------
package main

import "fmt"

func printBoard(board *[4][4]int) {
	for row := 0; row < 4; row++ {
		for col := 0; col < 4; col++ {
			if col > 0 {
				fmt.Print(" ")
			}
			fmt.Print(board[row][col])
		}
		fmt.Println()
	}
}

func main() {
	board := [4][4]int{
		{0, 0, 4, 0},
		{0, 3, 0, 0},
		{0, 0, 0, 2},
		{1, 0, 0, 0},
	}
	printBoard(&board)
}

ВЫВОД:
0 0 4 0
0 3 0 0
0 0 0 2
1 0 0 0

Те же два цикла, только теперь печатают не символы, а числа из массива, и нумерация идёт с нуля. Пробел ставится перед числом и только если col > 0 — так в конце строки не остаётся лишнего пробела. Ноль на экране означает пустую клетку.

ПРИМЕР 2.  Изменилось: доска попала в функцию — двумя способами
------------------------------------------------------------------------------
package main

import "fmt"

func fillCopy(board [2][2]int) {
	board[0][0] = 9
}

func fillReal(board *[2][2]int) {
	board[0][0] = 9
}

func main() {
	b := [2][2]int{{1, 2}, {3, 4}}

	fillCopy(b)
	fmt.Println(b)

	fillReal(&b)
	fmt.Println(b)
}

ВЫВОД:
[[1 2] [3 4]]
[[9 2] [3 4]]

Две функции отличаются одной звёздочкой, а результат — противоположный. fillCopy получила копию массива и изменила её; копия исчезла на выходе. fillReal получила адрес и изменила ту самую доску.
Сломай: убери * у fillReal и & при вызове — обе строки станут одинаковыми.

package main

import "fmt"

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}
	return true
}

func main() {
	board := [4][4]int{
		{0, 0, 4, 0},
		{0, 3, 0, 0},
		{0, 0, 0, 2},
		{1, 0, 0, 0},
	}
	fmt.Println(canPlace(&board, 0, 0, 1))
	fmt.Println(canPlace(&board, 0, 0, 2))
	fmt.Println(canPlace(&board, 0, 0, 3))
	fmt.Println(canPlace(&board, 0, 0, 4))
}

ВЫВОД:
false
true
true
false

Спрашиваем про клетку [0][0] четыре раза подряд. Единицу запрещает столбец (внизу стоит 1), четвёрку — строка (справа стоит 4), двойка и тройка проходят. Один цикл по k обходит и строку, и столбец: в первой проверке k — второй индекс, во второй — первый.

ПРИМЕР 4.  Изменилось: добавился квадрат 2×2 — одна вставка

package main

import "fmt"

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}

	startRow := row / 2 * 2
	startCol := col / 2 * 2
	for i := startRow; i < startRow+2; i++ {
		for j := startCol; j < startCol+2; j++ {
			if board[i][j] == val {
				return false
			}
		}
	}
	return true
}

func main() {
	board := [4][4]int{
		{0, 0, 4, 0},
		{0, 3, 0, 0},
		{0, 0, 0, 2},
		{1, 0, 0, 0},
	}
	fmt.Println(canPlace(&board, 0, 0, 1))
	fmt.Println(canPlace(&board, 0, 0, 2))
	fmt.Println(canPlace(&board, 0, 0, 3))
	fmt.Println(canPlace(&board, 0, 0, 4))
}

ВЫВОД:
false
true
false
false

Изменился ровно один ответ: тройка была разрешена, стала запрещена. Строка и столбец её пропускали, а квадрат — нет: тройка стоит в клетке [1][1], то есть в том же квадрате 2×2. row / 2 * 2 находит начало квадрата — на доске 4×4 делят на 2, на доске 9×9 делили бы на 3.

ПРИМЕР 5.  Изменилось: всё вместе: разбор, проверка, откат, печать
------------------------------------------------------------------------------
package main

import "fmt"

func main() {
	run([]string{"..4.", ".3..", "...2", "1..."})
	run([]string{"1234", "3412", "2143", "4323"})
}

func run(input []string) {
	var board [4][4]int
	for row := 0; row < 4; row++ {
		line := input[row]
		if len(line) != 4 {
			fmt.Println("Error")
			return
		}
		for col := 0; col < 4; col++ {
			sym := line[col]
			if sym == '.' {
				board[row][col] = 0
			} else if sym >= '1' && sym <= '4' {
				board[row][col] = int(sym - '0')
			} else {
				fmt.Println("Error")
				return
			}
		}
	}

	if !isValid(&board) {
		fmt.Println("Error")
		return
	}
	if !solve(&board) {
		fmt.Println("Error")
		return
	}
	printBoard(&board)
}

func isValid(board *[4][4]int) bool {
	for row := 0; row < 4; row++ {
		for col := 0; col < 4; col++ {
			val := board[row][col]
			if val == 0 {
				continue
			}
			board[row][col] = 0
			ok := canPlace(board, row, col, val)
			board[row][col] = val
			if !ok {
				return false
			}
		}
	}
	return true
}

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}
	startRow := row / 2 * 2
	startCol := col / 2 * 2
	for i := startRow; i < startRow+2; i++ {
		for j := startCol; j < startCol+2; j++ {
			if board[i][j] == val {
				return false
			}
		}
	}
	return true
}

func solve(board *[4][4]int) bool {
	for row := 0; row < 4; row++ {
		for col := 0; col < 4; col++ {
			if board[row][col] != 0 {
				continue
			}
			for val := 1; val <= 4; val++ {
				if canPlace(board, row, col, val) {
					board[row][col] = val
					if solve(board) {
						return true
					}
					board[row][col] = 0
				}
			}
			return false
		}
	}
	return true
}

func printBoard(board *[4][4]int) {
	for row := 0; row < 4; row++ {
		for col := 0; col < 4; col++ {
			if col > 0 {
				fmt.Print(" ")
			}
			fmt.Print(board[row][col])
		}
		fmt.Println()
	}
}

ВЫВОД:
2 1 4 3
4 3 2 1
3 4 1 2
1 2 3 4
Error

Программа целиком, только на доске 4×4 вместо 9×9. Первый вход решается и печатается. Второй заполнен полностью, но в последней строке две тройки — его останавливает isValid.
Сломай обязательно: замени if !isValid(&board) на if false и запусти снова. Вторая доска напечатается как есть, с двумя тройками: solve не нашёл ни одной пустой клетки и сразу сказал «решено».

ВЫВОД:

line[i] даёт байт, то есть число — код символа. У знака «1» он 49, у «0» — 48. Вычитание sym - '0' убирает сдвиг и оставляет саму цифру. Точка означает пустую клетку, для неё кладём ноль.

package main

import "fmt"

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}

	startRow := row / 2 * 2
	startCol := col / 2 * 2
	for i := startRow; i < startRow+2; i++ {
		for j := startCol; j < startCol+2; j++ {
			if board[i][j] == val {
				return false
			}
		}

func main() {
	board := [4][4]int{
		{2, 1, 4, 3},
		{4, 3, 0, 1},
		{3, 4, 1, 2},
		{1, 2, 3, 4},

	for val := 1; val <= 4; val++ {

Из четырёх цифр подошла одна — двойка. Мы её поставили, показали доску и вернули клетку в ноль. Вот эта последняя строка и есть откат: без неё неподошедшая цифра осталась бы лежать на доске и испортила все дальнейшие проверки.

package main

import "fmt"

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}

	startRow := row / 2 * 2
	startCol := col / 2 * 2
	for i := startRow; i < startRow+2; i++ {
		for j := startCol; j < startCol+2; j++ {
			if board[i][j] == val {
				return false
			}
		}
	}
	return true
}

func main() {
	board := [4][4]int{
		{2, 1, 4, 3},
		{4, 3, 2, 1},
		{3, 4, 1, 2},
		{1, 2, 3, 4},
	}
	val := board[0][0]
	fmt.Println(canPlace(&board, 0, 0, val))
	board[0][0] = 0
	fmt.Println(canPlace(&board, 0, 0, val))
	board[0][0] = val
	fmt.Println(board[0][0])
}

ВЫВОД:
false
true
2

Первый ответ false, хотя доска верная. Причина: canPlace ищет двойку по всей строке и находит её же саму — клетка ведь занята. Поэтому перед вопросом клетку временно обнуляют, а после ответа возвращают значение назад. Ровно так работает isValidBoard.
