ПОДГОТОВКА
==============================================================================

Создай папку темы и модуль — набери руками, не копируй:

    mkdir args1
    cd args1
    go mod init args1

Дальше весь код набирай в файл main.go в этой папке.
Запуск программы:

    go run .

------------------------------------------------------------------------------
КАК РАБОТАТЬ С ПРИМЕРАМИ
------------------------------------------------------------------------------

1. Набирай руками. Не копируй мышкой — пока печатаешь, глаз замечает
   скобки, точки и кавычки, которые при копировании проходят мимо.

2. ПЕРЕД запуском запиши, что программа выведет. Потом запусти и сверь
   с ответом под примером.

3. Не совпало — это и есть самое ценное место. Разберись, почему вышло
   именно так, и только потом иди дальше.

4. Заработало — сломай нарочно: убери кавычки, поменяй тип, поставь другой
   индекс. Прочитай, что скажет компилятор.

5. Каждый следующий пример отличается от предыдущего одним изменением —
   оно указано в заголовке. Найди его глазами до запуска.

------------------------------------------------------------------------------
ПРИМЕР 1.  Изменилось: первый, набирай целиком
------------------------------------------------------------------------------
package main

import (
	"fmt"
	"os"
)

func main() {
	fmt.Println(len(os.Args))
	fmt.Println(os.Args[1:])
}

ВЫВОД:
1
[]

Запустил без аргументов — а длина всё равно 1: внутри лежит путь к программе. Срез [1:] его отрезает, остаётся пустой слайс.
Запусти иначе: go run . 10 20 — увидишь 3 и [10 20].

------------------------------------------------------------------------------
ПРИМЕР 2.  Изменилось: аргументы прогнали циклом
------------------------------------------------------------------------------
package main

import (
	"fmt"
	"os"
)

func main() {
	args := os.Args[1:]
	for i, a := range args {
		fmt.Println(i, a)
	}
	fmt.Println("всего:", len(args))
}

ВЫВОД:
всего: 0

Цикл не сделал ни шага — слайс пустой, и это не ошибка. range по пустому слайсу просто ничего не выполняет.
Запусти иначе: go run . go is fun.

------------------------------------------------------------------------------
ПРИМЕР 3.  Изменилось: добавилась проверка на пустой список
------------------------------------------------------------------------------
package main

import (
	"fmt"
	"os"
)

func main() {
	args := os.Args[1:]
	if len(args) == 0 {
		fmt.Println("аргументов нет")
		return
	}
	fmt.Println("первый:", args[0])
}

ВЫВОД:
аргументов нет

Проверка стоит до обращения к args[0] — только поэтому программа не падает. Сломай: убери if целиком и запусти без аргументов — прочитай текст паники.

------------------------------------------------------------------------------
ПРИМЕР 4.  Изменилось: сообщение ушло в os.Stderr
------------------------------------------------------------------------------
package main

import (
	"fmt"
	"os"
)

func main() {
	args := os.Args[1:]
	if len(args) == 0 {
		fmt.Fprintln(os.Stderr, "аргументов нет")
		return
	}
	fmt.Println("первый:", args[0])
}

ВЫВОД:
аргументов нет

Вывод на экране тот же самый, что и в прошлом примере, — но поток другой. Проверь разницу: запусти оба примера как go run . > out.txt. Прошлый смолчит, этот покажет сообщение на экране.

------------------------------------------------------------------------------
ПРИМЕР 5.  Изменилось: return заменили на os.Exit
------------------------------------------------------------------------------
package main

import (
	"fmt"
	"os"
)

func main() {
	defer fmt.Println("это конец main")
	args := os.Args[1:]
	if len(args) == 0 {
		fmt.Println("аргументов нет, выходим")
		os.Exit(0)
	}
	fmt.Println("первый:", args[0])
}

ВЫВОД:
аргументов нет, выходим

Строка из defer не напечаталась: os.Exit обрывает программу немедленно, отложенные вызовы не выполняются. Сломай: замени os.Exit(0) на return — строка появится.

------------------------------------------------------------------------------
ПРИМЕР 6.  Изменилось: аргумент превращаем в число
------------------------------------------------------------------------------
package main

import (
	"fmt"
	"os"
	"strconv"
)

func main() {
	s := "42"
	if len(os.Args) > 1 {
		s = os.Args[1]
	}
	n, err := strconv.Atoi(s)
	if err != nil {
		fmt.Fprintln(os.Stderr, "не число:", s)
		return
	}
	fmt.Println(n * 2)
}

ВЫВОД:
84

Аргумента нет, поэтому взято значение по умолчанию "42" — обычный приём, когда параметр необязательный. Atoi сделал из строки число, и умножение стало возможным.
Запусти иначе: go run . 7 даст 14, а go run . abc — сообщение об ошибке.

------------------------------------------------------------------------------
ПРИМЕР 7.  Изменилось: значение по умолчанию стало нечисловым
------------------------------------------------------------------------------
package main

import (
	"fmt"
	"os"
	"strconv"
)

func main() {
	s := "42x"
	if len(os.Args) > 1 {
		s = os.Args[1]
	}
	n, err := strconv.Atoi(s)
	if err != nil {
		fmt.Println("n =", n)
		fmt.Println("err =", err)
		return
	}
	fmt.Println(n * 2)
}

ВЫВОД:
n = 0
err = strconv.Atoi: parsing "42x": invalid syntax

Два вывода сразу. Первый: при ошибке число не мусорное, а нулевое — и если ошибку не проверить, программа спокойно посчитает 0 * 2. Второй: err печатается напрямую и сам говорит, что именно не разобралось.

------------------------------------------------------------------------------
ПРИМЕР 8.  Изменилось: вместо аргумента — файл
------------------------------------------------------------------------------
package main

import (
	"fmt"
	"os"
)

func main() {
	err := os.WriteFile("note.txt", []byte("one\ntwo\nthree\n"), 0644)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		return
	}
	data, err := os.ReadFile("note.txt")
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		return
	}
	fmt.Println(len(data))
	fmt.Print(string(data))
}

ВЫВОД:
14
one
two
three

Программа сама создала файл рядом с собой, затем прочитала его. 14 — это байты вместе с тремя переводами строк, а не количество строк. В конце Print, а не Println: последний перевод строки уже в файле.
Сломай: замени fmt.Print(string(data)) на fmt.Println(data) и посмотри, что на самом деле лежит в data.

------------------------------------------------------------------------------
ПРИМЕР 9.  Изменилось: файла нет
------------------------------------------------------------------------------
package main

import (
	"fmt"
	"os"
)

func main() {
	data, err := os.ReadFile("missing.txt")
	if err != nil {
		fmt.Fprintln(os.Stderr, "не прочитал:", err)
		fmt.Fprintln(os.Stderr, "байтов получено:", len(data))
		return
	}
	fmt.Print(string(data))
}

ВЫВОД:
не прочитал: open missing.txt: no such file or directory
байтов получено: 0

Текст ошибки называет и файл, и причину — читать его полезнее, чем гадать. Данных при этом ноль: полагаться на data после ошибки нельзя.
Сломай: напиши data, _ := os.ReadFile(...) и убери if — программа промолчит и напечатает пустоту.

------------------------------------------------------------------------------
ПРИМЕР 10.  Изменилось: источник данных — stdin
------------------------------------------------------------------------------
package main

import (
	"bufio"
	"fmt"
	"os"
)

func main() {
	sc := bufio.NewScanner(os.Stdin)
	n := 0
	for sc.Scan() {
		n++
		fmt.Println(n, sc.Text())
	}
	fmt.Println("строк:", n)
}

ВЫВОД:
строк: 0

Такой вывод получится, если данных на входе нет. Запустишь просто go run . — программа будет ждать, пока ты печатаешь; нажми Ctrl+D, и она закончит.
Запусти иначе: printf 'one\ntwo\n' | go run . даст три строки, последняя — строк: 2.
