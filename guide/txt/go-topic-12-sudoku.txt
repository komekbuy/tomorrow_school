To run the program:

    go run .

------------------------------------------------------------------------------
HOW TO WORK WITH THE EXAMPLES
------------------------------------------------------------------------------

1. Type it out. Don't copy-paste — while you type, your eye catches the
   brackets, dots and quotes that copying slides right past.

2. BEFORE running, write down what it will print. Then run it and compare
   with the answer under the example.

3. A mismatch is the valuable part. Work out why it came out that way
   before moving on.

4. Once it works, break it on purpose: drop the quotes, change the type,
   use a different index. Read what the compiler says.

5. Each example differs from the previous one by a single change — it is
   named in the heading. Spot it before you run anything.

EXAMPLE 1.  What changed: shapes are done, the board begins
------------------------------------------------------------------------------
package main

import "fmt"

func printBoard(board *[4][4]int) {
	for row := 0; row < 4; row++ {
		for col := 0; col < 4; col++ {
			if col > 0 {
				fmt.Print(" ")
			}
			fmt.Print(board[row][col])
		}
		fmt.Println()
	}
}

func main() {
	board := [4][4]int{
		{0, 0, 4, 0},
		{0, 3, 0, 0},
		{0, 0, 0, 2},
		{1, 0, 0, 0},
	}
	printBoard(&board)
}

OUTPUT:
0 0 4 0
0 3 0 0
0 0 0 2
1 0 0 0

The same two loops, except now they print numbers from an array instead of characters, and the numbering starts at zero. The space goes before the number and only when col > 0, so no stray space is left at the end of a line. A zero on screen means an empty cell.

EXAMPLE 2.  What changed: the board goes into a function — two ways
------------------------------------------------------------------------------
package main

import "fmt"

func fillCopy(board [2][2]int) {
	board[0][0] = 9
}

func fillReal(board *[2][2]int) {
	board[0][0] = 9
}

func main() {
	b := [2][2]int{{1, 2}, {3, 4}}

	fillCopy(b)
	fmt.Println(b)

	fillReal(&b)
	fmt.Println(b)
}

OUTPUT:
[[1 2] [3 4]]
[[9 2] [3 4]]

The two functions differ by one star and give opposite results. fillCopy got a copy of the array and changed it; the copy died at the exit. fillReal got an address and changed the board itself.
Break it: drop the * on fillReal and the & at the call — both lines will read the same.

EXAMPLE 3.  What changed: the first rule of the game: no repeat in the row or the column
------------------------------------------------------------------------------
package main

import "fmt"

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}
	return true
}

func main() {
	board := [4][4]int{
		{0, 0, 4, 0},
		{0, 3, 0, 0},
		{0, 0, 0, 2},
		{1, 0, 0, 0},
	}
	fmt.Println(canPlace(&board, 0, 0, 1))
	fmt.Println(canPlace(&board, 0, 0, 2))
	fmt.Println(canPlace(&board, 0, 0, 3))
	fmt.Println(canPlace(&board, 0, 0, 4))
}

OUTPUT:
false
true
true
false

We ask about cell [0][0] four times in a row. The 1 is blocked by the column (there is a 1 at the bottom), the 4 by the row (there is a 4 to the right); 2 and 3 pass. One loop over k walks both the row and the column: in the first check k is the second index, in the second check the first.

EXAMPLE 4.  What changed: the 2×2 square is added — a single insert
------------------------------------------------------------------------------
package main

import "fmt"

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}

	startRow := row / 2 * 2
	startCol := col / 2 * 2
	for i := startRow; i < startRow+2; i++ {
		for j := startCol; j < startCol+2; j++ {
			if board[i][j] == val {
				return false
			}
		}
	}
	return true
}

func main() {
	board := [4][4]int{
		{0, 0, 4, 0},
		{0, 3, 0, 0},
		{0, 0, 0, 2},
		{1, 0, 0, 0},
	}
	fmt.Println(canPlace(&board, 0, 0, 1))
	fmt.Println(canPlace(&board, 0, 0, 2))
	fmt.Println(canPlace(&board, 0, 0, 3))
	fmt.Println(canPlace(&board, 0, 0, 4))
}

OUTPUT:
false
true
false
false

Exactly one answer changed: the 3 was allowed and is now refused. The row and the column let it through, the square did not — the 3 sits in cell [1][1], the same 2×2 square. row / 2 * 2 finds the start of the square — on a 4×4 board you divide by 2, on a 9×9 board you would divide by 3.

EXAMPLE 5.  What changed: all together: parsing, checking, backtracking, printing
------------------------------------------------------------------------------
package main

import "fmt"

func main() {
	run([]string{"..4.", ".3..", "...2", "1..."})
	run([]string{"1234", "3412", "2143", "4323"})
}

func run(input []string) {
	var board [4][4]int
	for row := 0; row < 4; row++ {
		line := input[row]
		if len(line) != 4 {
			fmt.Println("Error")
			return
		}
		for col := 0; col < 4; col++ {
			sym := line[col]
			if sym == '.' {
				board[row][col] = 0
			} else if sym >= '1' && sym <= '4' {
				board[row][col] = int(sym - '0')
			} else {
				fmt.Println("Error")
				return
			}
		}
	}

	if !isValid(&board) {
		fmt.Println("Error")
		return
	}
	if !solve(&board) {
		fmt.Println("Error")
		return
	}
	printBoard(&board)
}

func isValid(board *[4][4]int) bool {
	for row := 0; row < 4; row++ {
		for col := 0; col < 4; col++ {
			val := board[row][col]
			if val == 0 {
				continue
			}
			board[row][col] = 0
			ok := canPlace(board, row, col, val)
			board[row][col] = val
			if !ok {
				return false
			}
		}
	}
	return true
}

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}
	startRow := row / 2 * 2
	startCol := col / 2 * 2
	for i := startRow; i < startRow+2; i++ {
		for j := startCol; j < startCol+2; j++ {
			if board[i][j] == val {
				return false
			}
		}
	}
	return true
}

func solve(board *[4][4]int) bool {
	for row := 0; row < 4; row++ {
		for col := 0; col < 4; col++ {
			if board[row][col] != 0 {
				continue
			}
			for val := 1; val <= 4; val++ {
				if canPlace(board, row, col, val) {
					board[row][col] = val
					if solve(board) {
						return true
					}
					board[row][col] = 0
				}
			}
			return false
		}
	}
	return true
}

func printBoard(board *[4][4]int) {
	for row := 0; row < 4; row++ {
		for col := 0; col < 4; col++ {
			if col > 0 {
				fmt.Print(" ")
			}
			fmt.Print(board[row][col])
		}
		fmt.Println()
	}
}

OUTPUT:
2 1 4 3
4 3 2 1
3 4 1 2
1 2 3 4
Error

The whole program, just on a 4×4 board instead of 9×9. The first input is solved and printed. The second is completely filled but has two 3s in the last row — isValid stops it.
Break it, definitely: replace if !isValid(&board) with if false and run it again. The second board prints as is, two 3s and all: solve found no empty cell and declared it solved on the spot.

EXAMPLE 6.  What changed: a character turned into a digit
------------------------------------------------------------------------------
package main

import "fmt"

func main() {
	line := "1.4"
	for i := 0; i < len(line); i++ {
		sym := line[i]
		if sym == '.' {
			fmt.Println(0)
		} else {
			fmt.Println(int(sym - '0'))
		}
	}
}

OUTPUT:
1
0
4

line[i] gives a byte, that is a number — a character code. For "1" it is 49, for "0" it is 48. Subtracting sym - '0' removes the offset and leaves the digit itself. A dot means an empty cell, so we store a zero.

EXAMPLE 7.  What changed: we found the first empty cell
------------------------------------------------------------------------------
package main

import "fmt"

func main() {
	board := [4][4]int{
		{2, 1, 4, 3},
		{4, 3, 0, 1},
		{3, 4, 1, 2},
		{1, 2, 3, 4},
	}
	for row := 0; row < 4; row++ {
		for col := 0; col < 4; col++ {
			if board[row][col] == 0 {
				fmt.Println(row, col)
				return
			}
		}
	}
	fmt.Println("full")
}

OUTPUT:
1 2

The same walk as the frame: rows outside, columns inside. But we leave the moment we find one — return inside the loop. This is exactly how the solver finds where to put the next digit.

EXAMPLE 8.  What changed: a division and a multiplication in a row give the square's start
------------------------------------------------------------------------------
package main

import "fmt"

func main() {
	for i := 0; i < 4; i++ {
		fmt.Println(i, i/2*2)
	}
}

OUTPUT:
0 0
1 0
2 2
3 2

Integer division drops the remainder and the multiplication brings the scale back — but without it. The result is "round down to a multiple of two": rows 0 and 1 give 0, rows 2 and 3 give 2. That is the row a square starts at.
Break it: write i/(2*2) and compare — the brackets change the meaning entirely.

EXAMPLE 9.  What changed: we placed a digit and erased it right away
------------------------------------------------------------------------------
package main

import "fmt"

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}

	startRow := row / 2 * 2
	startCol := col / 2 * 2
	for i := startRow; i < startRow+2; i++ {
		for j := startCol; j < startCol+2; j++ {
			if board[i][j] == val {
				return false
			}
		}
	}
	return true
}

func main() {
	board := [4][4]int{
		{2, 1, 4, 3},
		{4, 3, 0, 1},
		{3, 4, 1, 2},
		{1, 2, 3, 4},
	}
	for val := 1; val <= 4; val++ {
		if canPlace(&board, 1, 2, val) {
			board[1][2] = val
			fmt.Println(val, board[1][2])
			board[1][2] = 0
			fmt.Println(val, board[1][2])
		}
	}
}

OUTPUT:
2 2
2 0

Out of four digits only one fitted — the two. We placed it, showed the board and reset the cell to zero. That last line is the backtrack: without it a digit that did not work out would stay on the board and spoil every later check.

EXAMPLE 10.  What changed: a cell is removed so that it can be asked about itself
------------------------------------------------------------------------------
package main

import "fmt"

func canPlace(board *[4][4]int, row, col, val int) bool {
	for k := 0; k < 4; k++ {
		if board[row][k] == val {
			return false
		}
		if board[k][col] == val {
			return false
		}
	}

	startRow := row / 2 * 2
	startCol := col / 2 * 2
	for i := startRow; i < startRow+2; i++ {
		for j := startCol; j < startCol+2; j++ {
			if board[i][j] == val {
				return false
			}
		}
	}
	return true
}

func main() {
	board := [4][4]int{
		{2, 1, 4, 3},
		{4, 3, 2, 1},
		{3, 4, 1, 2},
		{1, 2, 3, 4},
	}
	val := board[0][0]
	fmt.Println(canPlace(&board, 0, 0, val))
	board[0][0] = 0
	fmt.Println(canPlace(&board, 0, 0, val))
	board[0][0] = val
	fmt.Println(board[0][0])
}

OUTPUT:
false
true
2

The first answer is false, although the board is correct. The reason: canPlace looks for a two along the whole row and finds this very cell — it is occupied, after all. So before asking, the cell is temporarily zeroed and the value is put back afterwards. That is exactly how isValidBoard works.
